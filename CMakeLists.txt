cmake_minimum_required(VERSION 3.22)
project(CurvePilotAutonome VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_gui_app(CurvePilotAutonome PRODUCT_NAME "Curve Pilot Autonome 0.1" COMPANY_NAME ANKS)
juce_generate_juce_header(CurvePilotAutonome)
target_sources(CurvePilotAutonome PRIVATE Source/Main.cpp Source/Engine.h)
target_compile_definitions(CurvePilotAutonome PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(CurvePilotAutonome PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
  target_compile_options(CurvePilotAutonome PRIVATE /utf-8)
endif()
include(CTest)
add_executable(engine_tests tests/engine_tests.cpp)
target_include_directories(engine_tests PRIVATE tests/support)
target_compile_definitions(engine_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(engine_tests PRIVATE juce::juce_audio_utils)
add_test(NAME full_audio_buffers COMMAND engine_tests)
